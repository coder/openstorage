# Production-shape starvation probe — real S3, eu-central-1, 2026-08-27
# Serve: codestorage 52ea5c02 + PR #180 fix; bucket codestorage-q1ab-s3wal-7715761f
# (s3wal://…?region=eu-central-1 + s3://…?region=eu-central-1, expected-owner set)
# Invocation: ws8-fairness.sh 8091 hot 60 8
#   = 8 concurrent workers, one repo group ('hot'), 60 s window, each worker:
#   clone once, then loop { commit --allow-empty; time git push origin w<i> }.
#   Gap accounting per worker: last=worker start; on win, gap=now-last (so the
#   window-start->first-success gap is included); after the window, the trailing
#   last-success->window-end gap is folded into maxgap. Rejected pushes
#   (non-zero git push exit) increment rejects, then fetch+reset and retry.

## Method caveats (annotated post-hoc during review; raw sections below are verbatim)
# 1. Window boundary: workers test the deadline BEFORE starting a push, so a
#    push in flight at the deadline runs to completion unbounded — the nominal
#    60 s window can stretch by up to one push per worker (pushes ran ~1 s
#    here; the whole probe completed in ~64.6 s wall time), and a win landing
#    after the deadline is still counted. Direction of error for the
#    starvation evidence: conservative — the trailing gap is measured to the
#    worker's ACTUAL end, which is at or after the fixed deadline, so
#    reported gaps can only be equal or larger than deadline-clamped ones.
#    The aggregate wins/s divides by the nominal 60 s and may overstate the
#    rate by the overrun fraction; it is not used as gate evidence (the
#    throughput arm is measured by the §3 microbench cell).
# 2. No per-push timeout: a pathologically stalled push would have hung the
#    probe rather than being cut at the deadline. None occurred (the one
#    TLS-timeout rejection below was git's own client-side timeout). The
#    §2.3-conformant harness (landed as PR #184, closing #181, after this
#    probe was recorded) supersedes this probe; its real-S3 re-run should
#    bound pushes to the remaining window.

## Probe script (verbatim)
#!/usr/bin/env bash
set -euo pipefail
PORT=$1; REPO=$2; WINDOW=$3; WORKERS=$4
BIN=/tmp/ws8-bin/codestorage
unset GIT_ASKPASS GIT_SSH_COMMAND
export GIT_TERMINAL_PROMPT=0
TOK=$($BIN mint --key /tmp/ws8-key.pem --iss qual --repo "$REPO" --scopes git:read,git:write)
WORK=$(mktemp -d /tmp/ws8-fair-XXXX)
echo "work=$WORK"
for i in $(seq 1 $WORKERS); do
  git clone -q "http://t:${TOK}@127.0.0.1:$PORT/$REPO.git" "$WORK/w$i" 2>/dev/null
done
DEADLINE=$(( $(date +%s) + WINDOW ))
for i in $(seq 1 $WORKERS); do
(
  cd "$WORK/w$i"
  git checkout -qb "w$i"
  last=$(date +%s%N); maxgap=0; wins=0; rejects=0
  while [ "$(date +%s)" -lt "$DEADLINE" ]; do
    git -c user.name=q -c user.email=q@example.com commit -q --allow-empty -m "w$i $wins"
    if git push -q origin "w$i" 2>>"$WORK/rej-$i.log"; then
      now=$(date +%s%N); gap=$(( (now - last) / 1000000 ))
      [ "$gap" -gt "$maxgap" ] && maxgap=$gap
      last=$now; wins=$((wins+1))
    else
      rejects=$((rejects+1))
      git fetch -q origin "w$i" 2>/dev/null || true
      git reset -q --hard "origin/w$i" 2>/dev/null || true
    fi
  done
  end=$(date +%s%N); tail=$(( (end - last) / 1000000 ))
  [ "$tail" -gt "$maxgap" ] && maxgap=$tail
  echo "$i wins=$wins rejects=$rejects maxgap_ms=$maxgap" >> "$WORK/results.txt"
) &
done
wait
sort -n "$WORK/results.txt"
awk '{gsub(/wins=/,"",$2); gsub(/rejects=/,"",$3); gsub(/maxgap_ms=/,"",$4); w+=$2; r+=$3; if ($4+0>m) m=$4} END {printf "TOTAL wins=%d rejects=%d aggregate=%.1f/s worst_gap_ms=%d\n", w, r, w/'"$WINDOW"', m}' "$WORK/results.txt"
